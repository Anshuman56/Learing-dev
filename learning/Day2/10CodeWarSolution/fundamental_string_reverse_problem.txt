/*Complete the solution so that it reverses the string passed into it.

'world'  =>  'dlrow'
'word'   =>  'drow'*/

function solution(str){
  let wordArray = str.split('');
  let reverseArray = [];
  for ( let i = wordArray.length -1; i >= 0; i--){
    reverseArray.push(wordArray[i])
  }
  return reverseArray.join('')
}
